-- -----------------------------------------------------
-- Table : admin
-- Description : Gestion des administrateurs de la plateforme
-- -----------------------------------------------------
DROP TABLE IF EXISTS admin;
CREATE TABLE admin (
    id_adm INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identifiant unique de l\'administrateur',
    email_adm VARCHAR(255) NOT NULL UNIQUE COMMENT 'Email de l\'administrateur',
    mdp_adm VARCHAR(255) NOT NULL COMMENT 'Mot de passe hashé de l\'administrateur',
    role_adm ENUM('administrateur', 'super_root') DEFAULT 'administrateur' COMMENT 'Rôle de l\'administrateur',
    date_creation_adm TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT 'Date de création du compte admin',
    date_modification_adm TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Date de dernière modification'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='Table des administrateurs';

-- -----------------------------------------------------
-- Table : superviseur
-- Description : Gestion des superviseurs de la plateforme
-- -----------------------------------------------------
DROP TABLE IF EXISTS superviseur;
CREATE TABLE superviseur (
    id_sup INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identifiant unique du superviseur',
    email_sup VARCHAR(255) NOT NULL UNIQUE COMMENT 'Email du superviseur',
    mdp_sup VARCHAR(255) NOT NULL COMMENT 'Mot de passe hashé du superviseur',
    pseudo_sup varchar(50) DEFAULT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='Table des superviseurs';
    
-- -----------------------------------------------------
-- Table : utilisateurs
-- Description : Gestion des utilisateurs normaux de la plateforme
-- -----------------------------------------------------
DROP TABLE IF EXISTS utilisateurs;
CREATE TABLE utilisateurs (
    id_uti INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identifiant unique de l\'utilisateur',
    identifiant_etu VARCHAR(100) UNIQUE COMMENT 'Identifiant de l\'étudiant (pour CBS)',
    prenom_etu VARCHAR(100) COMMENT 'Prénom de l\'étudiant',
    nom_complet VARCHAR(255) COMMENT 'Nom complet de l\'utilisateur',
    email_uti VARCHAR(255) NOT NULL UNIQUE COMMENT 'Email de l\'utilisateur',
    mdp_uti VARCHAR(255) NOT NULL COMMENT 'Mot de passe hashé de l\'utilisateur',
    confirmation_mdp_uti VARCHAR(255) COMMENT 'Token de confirmation du mot de passe',
    confirmation_compte_uti TIMESTAMP NULL COMMENT 'Date de confirmation du compte (NULL si non confirmé)',
    reset_token VARCHAR(255) COMMENT 'Token pour réinitialisation du mot de passe',
    reset_at TIMESTAMP NULL COMMENT 'Date de demande de réinitialisation',
    connexion_auto VARCHAR(255) COMMENT 'Token pour connexion automatique',
    image_uti VARCHAR(500) COMMENT 'Chemin vers l\'image de profil',
    date_creation_uti TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT 'Date de création du compte',
    date_modification_uti TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Date de dernière modification'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='Table des utilisateurs';

-- -----------------------------------------------------
-- Table : etudiant_cbs
-- Description : Étudiants CBS confirmés et actifs
-- -----------------------------------------------------
DROP TABLE IF EXISTS etudiant_cbs;
CREATE TABLE etudiant_cbs (

    prenom_etu VARCHAR(100) NOT NULL COMMENT 'Prénom de l\'étudiant',

    carte_verso_etu VARCHAR(500) COMMENT 'Chemin vers l\'image verso de la carte',
    date_inscription_etu TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT 'Date d\'inscription',
    date_confirmation_etu TIMESTAMP NULL COMMENT 'Date de confirmation du compte',
    date_modification_etu TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Date de dernière modification'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='Table des étudiants CBS confirmés';

-- -----------------------------------------------------
-- Table : etudiant_cbs_temp
-- Description : Demandes d'inscription CBS en attente de validation
-- -----------------------------------------------------
DROP TABLE IF EXISTS etudiant_cbs_temp;
CREATE TABLE etudiant_cbs_temp (
    id_etu INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identifiant unique de la demande',
    prenom_etu VARCHAR(100) NOT NULL COMMENT 'Prénom de l\'étudiant',
    email_etu VARCHAR(255) NOT NULL COMMENT 'Email de l\'étudiant',
    carte_recto_etu VARCHAR(500) COMMENT 'Chemin vers l\'image recto de la carte',
    carte_verso_etu VARCHAR(500) COMMENT 'Chemin vers l\'image verso de la carte',
    compte_confirme_etu TINYINT(1) DEFAULT 0, -- 0 = non confirmé, 1 = confirmé
    identifiant_etu VARCHAR(100) COMMENT 'Identifiant généré pour l\'étudiant',
    date_inscription_etu TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT 'Date de la demande d\'inscription',
    statut_demande ENUM('en_attente', 'approuvee', 'rejetee') DEFAULT 'en_attente' COMMENT 'Statut de la demande'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='Table des demandes d\'inscription CBS en attente';
